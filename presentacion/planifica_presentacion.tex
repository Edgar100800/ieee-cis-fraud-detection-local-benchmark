% ---------- 6 · Protocolo adaptativo ----------
\begin{frame}{Cada bloque futuro usa solo la información disponible}
  \framesubtitle{El entrenamiento y las agregaciones respetan el instante de cada transacción}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \begin{beamercolorbox}[center,sep=0.22cm,rounded=true]{mist}%
        {\normalsize\bfseries\color{ink}60 días inicial \;$\rightarrow$\; bloques de 30 días}\\

      \begin{block}{Criterio de actualización}
        \small PSI alto o caída de desempeño $\rightarrow$ revisar ventana y reentrenar.

% ---------- 7 · Resultados adaptativos ----------
\begin{frame}{Reducir la historia no garantiza mejor rendimiento}
  \framesubtitle{Corrida completa: 590\,540 transacciones, bloques futuros de 30 días, XGBoost de 180 árboles}

    \begin{column}{0.36\textwidth}

\begin{frame}{Conclusiones}
  \framesubtitle{La línea base orienta la adaptación}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{\textcolor{white}{Features causales}}
        \small El UID aporta contexto de entidad sin usar información futura:
        $+0{,}01$ AUC de pura ingeniería de representación.
      \end{block}
      \begin{block}{\textcolor{white}{Ventanas comparables}}
        \small La ventana expansiva fue la mejor en esta corrida (AUC 0,9163 vs 0,9039 con 30 días).
      \end{block}
      \begin{block}{\textcolor{white}{Monitoreo operativo}}
        \small AUC, PSI y costo de decisión deciden cuándo reentrenar; el accuracy alto
